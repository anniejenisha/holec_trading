frappe.pages['supplier-list-view'].on_page_load = function(wrapper) { var page = frappe.ui.make_app_page({ parent: wrapper, title: 'Suppliers', single_column: true }); load_supplier_table(page); }; function load_supplier_table(page, search_term = '') { let filters = []; if (search_term) { filters = [ ['Supplier', 'supplier_name', 'like', `%${search_term}%`] ]; } frappe.call({ method: 'frappe.client.get_list', args: { doctype: 'Supplier', fields: ['name', 'supplier_name', 'supplier_group', 'county', 'tax_id', 'disabled', 'is_frozen'], filters: search_term ? [ ['Supplier', 'supplier_name', 'like', `%${search_term}%`] ] : [], or_filters: search_term ? [ ['Supplier', 'tax_id', 'like', `%${search_term}%`] ] : [], limit_page_length: 50, order_by: 'creation desc' }, callback: function(r) { let suppliers = r.message || []; render_table_html(page, suppliers, search_term); } }); } function render_table_html(page, suppliers, search_term) { let rows = suppliers.map(sup => { let status_badge = sup.disabled ? `Draft` : `Approved`; return ` ${sup.name} ${sup.supplier_name} ${sup.supplier_group || '-'} ${sup.county || '-'} ${sup.tax_id || '-'} ${status_badge} `; }).join(''); let html = `

Suppliers ${suppliers.length}

${rows.length ? rows : ''}
ID Name Group County KRA PIN Status
No suppliers found
`; $(page.body).html(html); // Event Delegation for row clicks to prevent bubbling issues $(page.body).find('tbody tr').on('click', function() { let supplier_name = $(this).attr('data-name'); frappe.set_route('supplier-details', supplier_name); }); // New Supplier Button Handler $(page.body).find('#btn-new-supplier').on('click', function() { frappe.new_doc('Supplier'); }); // Debounced Search Input Handler let timeout; $(page.body).find('#supplier-search').on('keyup', function() { clearTimeout(timeout); let val = $(this).val(); timeout = setTimeout(() => { load_supplier_table(page, val); }, 300); }); }
